/*
 *  Sends the test parameter structure and number of tests
 */
void AUDIOLIB_rmsSmoothed_getTestParams (
    AUDIOLIB_rmsSmoothed_testParams_t **params,
    int32_t *                 numTests)
{
   *params = testParams;
   *numTests =
       sizeof (testParams) / sizeof (AUDIOLIB_rmsSmoothed_testParams_t);
}
