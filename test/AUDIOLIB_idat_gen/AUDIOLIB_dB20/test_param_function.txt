/*
 *  Sends the test parameter structure and number of tests
 */
void AUDIOLIB_dB20_getTestParams (
    AUDIOLIB_dB20_testParams_t **params,
    int32_t *                                 numTests)
{
   *params = testParams;
   *numTests =
       sizeof (testParams) / sizeof (AUDIOLIB_dB20_testParams_t);
}
